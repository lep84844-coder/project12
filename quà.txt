    <script>
        // State variables
        let dodgeCount = 0;
        let currentFxMode = 'matrix';
        let effectSpeed = 1.0;
        let themeColor = '#ff0077';

        // --- DODGE LOGIC FOR "NO" BUTTON ---
        const noBtn = document.getElementById('noBtn');

        function dodgeNoButton(e) {
            if (e) e.preventDefault();
            dodgeCount++;

            if (dodgeCount > 8) {
                acceptLove();
                return;
            }

            const padding = 60;
            const maxX = window.innerWidth - noBtn.offsetWidth - padding;
            const maxY = window.innerHeight - noBtn.offsetHeight - padding;

            const newX = Math.max(padding, Math.random() * maxX);
            const newY = Math.max(padding, Math.random() * maxY);

            noBtn.style.position = 'fixed';
            noBtn.style.left = `${newX}px`;
            noBtn.style.top = `${newY}px`;

            if (dodgeCount === 3) noBtn.innerText = "Chắc chưa? 😜";
            if (dodgeCount === 5) noBtn.innerText = "Đố bắt được! 🚀";
            if (dodgeCount === 7) noBtn.innerText = "Chọn CÓ đi! 🥺";
        }

        function acceptLove() {
            document.getElementById('screen1').classList.add('hidden');
            document.getElementById('screen2').classList.remove('hidden');
            triggerHeartBurst();
        }

        function resetApp() {
            dodgeCount = 0;
            noBtn.style.position = 'relative';
            noBtn.style.left = 'auto';
            noBtn.style.top = 'auto';
            noBtn.innerText = "Không 😜";

            document.getElementById('screen2').classList.add('hidden');
            document.getElementById('screen1').classList.remove('hidden');
        }

        // --- SETTINGS PANEL TOGGLE ---
        function toggleSettingsPanel() {
            const panel = document.getElementById('settingsPanel');
            panel.classList.toggle('translate-x-full');
        }

        function updateQuestionText() {
            const val = document.getElementById('questionInput').value;
            document.getElementById('questionText').innerText = val || "Do You Love Me?";
        }

        // --- BACKGROUND IMAGE HANDLERS ---
        function setBgImage(url) {
            const bgDiv = document.getElementById('customBg');
            bgDiv.style.backgroundImage = `url('${url}')`;
        }

        function resetBackground() {
            document.getElementById('customBg').style.backgroundImage = 'none';
            document.getElementById('bgUrlInput').value = '';
        }

        function handleFileUpload(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(evt) {
                    setBgImage(evt.target.result);
                };
                reader.readAsDataURL(file);
            }
        }

        function applyBgUrl() {
            let url = document.getElementById('bgUrlInput').value.trim();
            if (!url) return;

            // Auto conversion for Google Drive view links
            if (url.includes('drive.google.com')) {
                const match = url.match(/\/d\/([a-zA-Z0-9_-]+)/);
                if (match && match[1]) {
                    url = `https://lh3.googleusercontent.com/d/${match[1]}`;
                }
            }
            setBgImage(url);
        }

        function updateDarkness(val) {
            document.getElementById('darknessVal').innerText = val + '%';
            document.getElementById('bgOverlay').style.backgroundColor = `rgba(0, 0, 0, ${val / 100})`;
        }

        function updateFxSpeed(val) {
            effectSpeed = parseFloat(val);
            document.getElementById('speedVal').innerText = effectSpeed.toFixed(1) + 'x';
        }

        function setThemeColor(hex) {
            themeColor = hex;
            document.documentElement.style.setProperty('--neon-color', hex);
            document.getElementById('pulseSvg').style.color = hex;
            document.getElementById('screen2Title').style.color = hex;
        }

        function changeFxMode() {
            currentFxMode = document.getElementById('fxModeSelect').value;
            initCanvasParticles();
        }

        // --- CANVAS DYNAMIC VISUAL EFFECTS ANIMATION SYSTEM ---
        const canvas = document.getElementById('fxCanvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            initCanvasParticles();
        }
        window.addEventListener('resize', resizeCanvas);

        function initCanvasParticles() {
            particles = [];
            const count = currentFxMode === 'matrix' ? 40 : 60;

            for (let i = 0; i < count; i++) {
                particles.push(createParticle());
            }
        }

        function createParticle(isBurst = false) {
            return {
                x: Math.random() * canvas.width,
                y: isBurst ? canvas.height / 2 : Math.random() * canvas.height,
                size: Math.random() * 18 + 12,
                vx: (Math.random() - 0.5) * (isBurst ? 8 : 2),
                vy: isBurst ? (Math.random() - 0.5) * 8 : (Math.random() * 1.5 + 0.5),
                alpha: Math.random() * 0.8 + 0.2,
                char: "LOVE YOU",
                rotation: Math.random() * Math.PI * 2,
                rotSpeed: (Math.random() - 0.5) * 0.05
            };
        }

        function triggerHeartBurst() {
            for (let i = 0; i < 35; i++) {
                particles.push(createParticle(true));
            }
        }

        // Render Loop
        function renderCanvas() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            if (currentFxMode === 'matrix') {
                // TIKTOK MATRIX RUNNING TEXT
                ctx.save();
                ctx.font = '12px "Press Start 2P"';
                ctx.fillStyle = themeColor;
                ctx.globalAlpha = 0.25;
                ctx.rotate(-0.2);

                const text = "LOVE YOU LOVE YOU ";
                const rows = Math.ceil(canvas.height / 24) + 10;
                const cols = Math.ceil(canvas.width / 180) + 5;
                
                const time = Date.now() * 0.05 * effectSpeed;

                for (let r = -5; r < rows; r++) {
                    for (let c = -5; c < cols; c++) {
                        const x = c * 180 + (r % 2 === 0 ? 30 : 0);
                        const y = r * 28 + (time % 28);
                        ctx.fillText(text, x - 100, y);
                    }
                }
                ctx.restore();

            } else if (currentFxMode === 'rain') {
                // HEART RAIN
                particles.forEach((p, idx) => {
                    p.y += p.vy * effectSpeed * 2;
                    if (p.y > canvas.height + 30) {
                        particles[idx] = createParticle();
                        particles[idx].y = -20;
                    }

                    ctx.save();
                    ctx.globalAlpha = p.alpha;
                    ctx.font = `${p.size}px sans-serif`;
                    ctx.fillStyle = themeColor;
                    ctx.fillText('💖', p.x, p.y);
                    ctx.restore();
                });

            } else if (currentFxMode === 'snow') {
                // HEART SNOW
                particles.forEach((p, idx) => {
                    p.y += p.vy * effectSpeed;
                    p.x += Math.sin(p.y * 0.02) * 1.5;
                    p.rotation += p.rotSpeed;

                    if (p.y > canvas.height + 30) {
                        particles[idx] = createParticle();
                        particles[idx].y = -20;
                    }

                    ctx.save();
                    ctx.globalAlpha = p.alpha;
                    ctx.translate(p.x, p.y);
                    ctx.rotate(p.rotation);
                    ctx.font = `${p.size}px sans-serif`;
                    ctx.fillText('❄️❤️', 0, 0);
                    ctx.restore();
                });

            } else if (currentFxMode === 'fireworks') {
                // FIREWORKS SPARKLES
                particles.forEach((p, idx) => {
                    p.x += p.vx * effectSpeed;
                    p.y += p.vy * effectSpeed;
                    p.alpha -= 0.008 * effectSpeed;

                    if (p.alpha <= 0) {
                        particles[idx] = createParticle(true);
                    }

                    ctx.save();
                    ctx.globalAlpha = Math.max(0, p.alpha);
                    ctx.font = `${p.size}px sans-serif`;
                    ctx.fillText('✨💗', p.x, p.y);
                    ctx.restore();
                });
            }

            requestAnimationFrame(renderCanvas);
        }

        // Initialize on Load
        window.onload = function() {
            resizeCanvas();
            renderCanvas();
        };
    </script>

<script>
const envelope = document.getElementById("envelope");
const openBtn = document.getElementById("openBtn");
const closeBtn = document.getElementById("closeBtn");
const hearts = document.getElementById("hearts");

let isOpen = false;

function openLetter() {
  if (isOpen) return;

  isOpen = true;
  envelope.classList.add("open");
  openBtn.textContent = "💖 Đã mở thư";

  createHearts(25);
}

function closeLetter() {
  isOpen = false;
  envelope.classList.remove("open");
  openBtn.textContent = "💌 Mở thư";
}

function createHearts(amount) {
  for (let i = 0; i < amount; i++) {
    const heart = document.createElement("span");

    heart.className = "floating-heart";
    heart.textContent = ["💗", "💖", "💕", "✨"][
      Math.floor(Math.random() * 4)
    ];

    heart.style.left = Math.random() * 100 + "vw";
    heart.style.fontSize = 15 + Math.random() * 25 + "px";
    heart.style.animationDuration = 3 + Math.random() * 3 + "s";

    hearts.appendChild(heart);

    heart.addEventListener("animationend", () => {
      heart.remove();
    });
  }
}

envelope.addEventListener("click", openLetter);
openBtn.addEventListener("click", openLetter);
closeBtn.addEventListener("click", closeLetter);

// Tạo trái tim bay liên tục
setInterval(() => {
  if (isOpen) createHearts(2);
}, 1000);
</script>
